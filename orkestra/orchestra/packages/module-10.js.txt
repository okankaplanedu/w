  function noteName(m) {
    var N = ['C', 'Db', 'D', 'Eb', 'E', 'F', 'Gb', 'G', 'Ab', 'A', 'Bb', 'B'];
    return N[((m % 12) + 12) % 12] + (Math.floor(m / 12) - 1);

  /* --- yükleme: bozuk/sessiz örnekleri ayıkla (contrabass C4 gibi) --- */
  ORC.loadInstrument = function (inst) {
    if (buffers[inst]) return Promise.resolve(buffers[inst]);
    if (pending[inst]) return pending[inst];
    var src = (root.ORC_SAMPLES && root.ORC_SAMPLES[inst]) || {};
    var keys = Object.keys(src);
    pending[inst] = Promise.all(keys.map(function (k) {
      var url = src[k];
      return fetch(url).then(function (r) { return r.arrayBuffer(); })
        .then(function (ab) {
          return new Promise(function (res) {
            ctx.decodeAudioData(ab, function (buf) {
              /* sessiz örnek kapısı */
              try {
                var d = buf.getChannelData(0), mx = 0;
                for (var i = 0; i < d.length; i += 64) { var v = Math.abs(d[i]); if (v > mx) mx = v; }
                if (mx < 0.004) return res(null);
              } catch (e) { }
              res({ midi: +k, buf: buf });
            }, function () { res(null); });
          });
        }).catch(function () { return null; });
    })).then(function (list) {
      var arr = list.filter(Boolean).sort(function (a, b) { return a.midi - b.midi; });
      buffers[inst] = arr;
      loadedCount++;
      return arr;
    });
    return pending[inst];
  };

  ORC.loadAll = function (onStep) {
    loading = true;
    var insts = Object.keys(CHAR).filter(function (i) { return root.ORC_SAMPLES && root.ORC_SAMPLES[i]; });
    totalInst = insts.length; loadedCount = 0;
    var chain = Promise.resolve();
    insts.forEach(function (inst) {
      chain = chain.then(function () {
        return ORC.loadInstrument(inst).then(function () {
          if (onStep) onStep(loadedCount, totalInst, inst);
        });
      });
    });
    return chain.then(function () { ready = true; loading = false; return true; });
  };

  /* döngü noktasını sıfır geçişine oturt (tık/çızırtı önleme) */
  function snapZero(buf, sec) {
    try {
      var sr = buf.sampleRate, d = buf.getChannelData(0);
      var i = Math.max(1, Math.min(d.length - 2, Math.floor(sec * sr)));
      var span = Math.floor(sr * 0.012);
      for (var k = 0; k < span; k++) {
        if (i + k < d.length - 2 && d[i + k] <= 0 && d[i + k + 1] > 0) return (i + k) / sr;
        if (i - k > 1 && d[i - k] <= 0 && d[i - k + 1] > 0) return (i - k) / sr;
      }
      return sec;
    } catch (e) { return sec; }
  }

  // Bu, kaydın kendi spektral ölçümü değil: nota perdesi, enstrüman ve
  // velocity üzerinden hesaplanan sınırlı bir parlaklık/risk tahminidir.
  function sharpRisk(inst, midi, vel) {
    var families = { violin:1, viola:1, string_ensemble_1:1, string_ensemble_2:1, tremolo_strings:1,
      flute:1, oboe:1, clarinet:1, trumpet:1, french_horn:1, brass_section:1, glockenspiel:1, tubular_bells:1 };
    if (!families[inst]) return 0;
    var pitch = Math.max(0, Math.min(1, (midi - 72) / 25));
    var accent = Math.max(0, Math.min(1, (vel - 0.40) / 0.80));
    return Math.max(0, Math.min(1, pitch * (0.65 + 0.35 * accent))) * masterConfig.softness;
  }

  /* --- ses çalma: tek nota, aile veri yoluna bağlanır --- */
  function voice(inst, ch, s, t0, hold, amp, vel, opts, f0) {
    var rate = Math.pow(2, (opts.midi - s.midi) / 12);
    // Örnek transpozisyonu gerçek perde oranını korur; 0.70–1.42 sıkıştırması detoneye yol açıyordu.
    rate = Math.max(0.125, Math.min(8, rate));

    var src = ctx.createBufferSource();
    src.buffer = s.buf;
    src.playbackRate.value = rate;

    var bd = s.buf.duration / rate;
    var useLoop = !ch.perc && ch.sus && hold > (bd * 0.62);
    if (useLoop && bd > 0.6) {
      try {
        src.loop = true;
        src.loopStart = snapZero(s.buf, s.buf.duration * 0.38);
        src.loopEnd = Math.max(src.loopStart + 0.12, snapZero(s.buf, s.buf.duration * 0.86));
      } catch (e) { }
    }

    var filt = ctx.createBiquadFilter();
    filt.type = 'lowpass';
    var comfortCut = ch.cut * (0.90 + 0.30 * Math.min(1.3, vel)) * (opts.cut || 1);
    // Nota başına yüksek frekans sınırı: perde değişmez, yalnız parlaklık azalır.
    if (opts.midi >= 88) comfortCut = Math.min(comfortCut, Math.max(5000, 8500 - (opts.midi - 88) * 110));
    var expressiveFam = FAMILY[inst] === 'str' || FAMILY[inst] === 'wood' || FAMILY[inst] === 'brass';
    var softRisk = sharpRisk(inst, opts.midi, vel);
    if (expressiveFam) comfortCut *= 1 - 0.18 * masterConfig.haze;
    comfortCut *= 1 - 0.36 * softRisk;
    if (expressiveFam && opts.midi >= 79) comfortCut = Math.min(comfortCut, 6200 - Math.min(1800, (opts.midi - 79) * 70));
    filt.frequency.value = clampF(comfortCut);
    filt.Q.value = 0.4;

    var hpN = ctx.createBiquadFilter();
    hpN.type = 'highpass';
    hpN.frequency.value = clampF(Math.max(30, Math.min(900, f0 * 0.55 * (ch.hp || 1))));
    hpN.Q.value = 0.5;

    var g = ctx.createGain();
    var atk = Math.max(0.002, ch.atk * (ch.perc ? 0.9 : (1.15 - 0.25 * Math.min(1.3, vel))));
    if (expressiveFam) atk *= 1 + 0.40 * softRisk;
    var rel = ch.perc ? ((opts.bridgeAttack || opts.harpGrain) ? Math.min(ch.rel, 0.22) : ch.rel) : (ch.rel * (0.9 + 0.4 * (1 - Math.min(1, vel))));
    var ampOk = (isFinite(amp) && amp > 0) ? amp : 0.02;

    g.gain.setValueAtTime(0.0001, Math.max(0, t0 - 0.004));
    g.gain.linearRampToValueAtTime(ampOk, t0 + atk);

    if (ch.perc && (opts.bridgeAttack || opts.harpGrain)) {
      // Kısa arpej notalarında kuyruklar akor değişimine taşmasın.
      g.gain.setTargetAtTime(0.0001, t0 + atk + Math.min(hold, opts.harpGrain ? 0.16 : 0.10), 0.065);
    } else if (ch.perc) {
      /* tok vuruş: uzun doğal kuyruk, yumuşak iniş */
      var tail = Math.max(0.25, rel);
      g.gain.setTargetAtTime(ampOk * 0.62, t0 + atk + 0.4, tail * 0.55);
      g.gain.setTargetAtTime(0.0001, t0 + atk + tail * 0.55, tail * 0.42);
    } else {
      var susEnd = t0 + Math.max(0.06, hold);
      g.gain.setValueAtTime(ampOk, Math.max(t0 + atk + 0.002, susEnd - rel * 0.35));
      g.gain.setTargetAtTime(0.0001, susEnd, rel * 0.34);
    }

    src.connect(hpN); hpN.connect(filt); filt.connect(g);

    var pan = ctx.createStereoPanner ? ctx.createStereoPanner() : null;
    var panV = (opts.px != null) ? (opts.px * 0.85 + ch.pan * 0.15) : (ch.pan + (opts.pan || 0));
    if (pan) pan.pan.value = Math.max(-1, Math.min(1, panV));

    var dist = (opts.pz != null) ? Math.abs(opts.pz) : 0;
    var fam = FAMILY[inst] || 'str';
    var bus = buses[fam] || buses.str;
    if (pan) { g.connect(pan); pan.connect(bus.g); } else { g.connect(bus.g); }

    var rg = ctx.createGain();
    rg.gain.value = ch.wet * (opts.wet != null ? opts.wet : 1) * (1 + dist * 0.8) * (expressiveFam ? 0.90 + 0.28 * masterConfig.haze + 0.10 * softRisk : 1);
    if (pan) pan.connect(rg); else g.connect(rg);
    rg.connect(reverb);

    /* eşzamanlı ses tavanı: en eski sesi yumuşak kıs (asla ani kesme) */
    if (voices >= VOICE_MAX) {
      var old = live[liveIdx % live.length];
      if (old) { try { old.gain.cancelScheduledValues(ctx.currentTime); old.gain.setTargetAtTime(0.0001, ctx.currentTime, 0.03); } catch (e2) { } }
      stolen++;
      voices = Math.max(0, voices - 1);
    }
    live[liveIdx % 512] = g; liveIdx++;
    voices++;
    src.onended = function () { voices = Math.max(0, voices - 1); };

    var playDur = (useLoop ? hold + rel * 2.2 + 0.5 : Math.min(bd, hold + rel * 2.2 + 0.25));
    playDur = Math.max(0.12, playDur);
    // Sürekli perdeli yaylı/üflemelilerde yumuşak, nota ömrüyle sınırlı vibrato.
    // Perde merkezi değişmez; her nota için derinlik doğal ataktan sonra açılır.
    if (expressiveFam && !ch.perc && ch.sus && hold >= 0.34 && voices < 96 && masterConfig.vibrato > 0) {
      try {
        var lfo = ctx.createOscillator(), vib = ctx.createGain();
        lfo.type = 'sine'; lfo.frequency.value = 4.5 + (opts.midi % 7) * 0.11;
        var cents = Math.min(8, masterConfig.vibrato * (20 + 6 * softRisk));
        var depth = rate * (Math.pow(2, cents / 1200) - 1);
        vib.gain.setValueAtTime(0, t0);
        vib.gain.setValueAtTime(0, t0 + Math.min(0.12, playDur * 0.18));
        vib.gain.linearRampToValueAtTime(depth, t0 + Math.min(0.38, playDur * 0.45));
        vib.gain.setTargetAtTime(0, t0 + Math.max(0.05, playDur - 0.15), 0.05);
        lfo.connect(vib); vib.connect(src.playbackRate);
        // Yüksek perdeli nefeslilerde aynı yumuşak titreşim, filtreye de az miktarda uygulanır.
        // Perde merkezini kaydırmaz, yeni ses katmanı oluşturmaz.
        var throat=null;
        if ((fam === 'wood' || fam === 'brass') && softRisk > 0.10) {
          throat=ctx.createGain();
          var depthHz=Math.min(180, filt.frequency.value * 0.028 * softRisk * masterConfig.vibrato / 0.34);
          throat.gain.setValueAtTime(0,t0);
          throat.gain.linearRampToValueAtTime(depthHz,t0+Math.min(0.35,playDur*0.4));
          throat.gain.setTargetAtTime(0,t0+Math.max(0.05,playDur-0.15),0.05);
          lfo.connect(throat);throat.connect(filt.frequency);
        }
        lfo.start(t0); lfo.stop(t0 + playDur + 0.06);
        lfo.onended = function(){ try { lfo.disconnect(); vib.disconnect(); if(throat)throat.disconnect(); } catch (ignore) {} };
      } catch (vibError) { /* vibrato desteklenmezse örnek normal çalar */ }
    }
    try {
      var off = Math.min(Math.max(0, s.buf.duration - 0.25), Math.random() * 0.012);
      src.start(t0, off, playDur);
      src.stop(t0 + playDur + 0.05);
    } catch (e) { try { src.start(t0); } catch (e2) { } }

    /* ışıltı katmanı */
    if (!masterConfig.economy && SHINE[inst] && vel > 0.45 && !ch.perc && voices < 70) {
      var src2 = ctx.createBufferSource();
      src2.buffer = s.buf; src2.playbackRate.value = rate * 2;
      var hp2 = ctx.createBiquadFilter();
      hp2.type = 'highpass'; hp2.frequency.value = clampF(Math.max(1400, f0 * 1.6)); hp2.Q.value = 0.6;
      var g2 = ctx.createGain();
      var a2 = ampOk * SHINE[inst] * 0.55;
      if (!isFinite(a2) || a2 <= 0) a2 = ampOk * 0.12;
      g2.gain.setValueAtTime(0.0001, t0);
      g2.gain.linearRampToValueAtTime(a2, t0 + Math.max(0.03, atk * 1.6));
      g2.gain.setTargetAtTime(0.0001, t0 + Math.max(0.12, hold * 0.7), 0.18);
      src2.connect(hp2); hp2.connect(g2);
      if (pan) g2.connect(pan); else g2.connect(bus.g);
      try { src2.start(t0, 0.01, Math.min(1.4, hold * 0.8 + 0.3)); src2.stop(t0 + hold * 0.8 + 0.4); } catch (e3) { }
    }
    return true;
  }

  /* vurmalı gövde desteği: alt oktav, düşük seviye → dijital değil, tok */
  var SUB = { timpani: 0.30, taiko_drum: 0.24, melodic_tom: 0.18, orchestra_hit: 0.20 };

  var fxNoise = null;
  function battleFx(inst, when, vel, opts) {
    if (!ctx || !ready || !masterConfig.battle) return false;
    var t = Math.max(when, ctx.currentTime + 0.004), duration = inst === 'battle_arrow' ? 0.22 : 0.30;
    if (!fxNoise) {
      fxNoise = ctx.createBuffer(1, Math.floor(ctx.sampleRate * 0.35), ctx.sampleRate);
      var data = fxNoise.getChannelData(0);
      for (var i=0;i<data.length;i++) data[i]=(Math.random()*2-1);
    }
    var noise = ctx.createBufferSource(); noise.buffer=fxNoise;
    var filter = ctx.createBiquadFilter(); filter.type='bandpass';
    filter.frequency.setValueAtTime(inst === 'battle_arrow' ? 1100 : 2300, t);
    filter.frequency.exponentialRampToValueAtTime(inst === 'battle_arrow' ? 3600 : 900, t+duration);
    filter.Q.value=inst === 'battle_arrow' ? 1.4 : 2.4;
    var g=ctx.createGain(), pan=ctx.createStereoPanner ? ctx.createStereoPanner() : null;
    g.gain.setValueAtTime(0.0001,t);g.gain.linearRampToValueAtTime(Math.min(0.09,vel*0.12),t+0.012);
    g.gain.exponentialRampToValueAtTime(0.0001,t+duration);
    noise.connect(filter);filter.connect(g);
    if(pan){pan.pan.value=Math.max(-0.6,Math.min(0.6,(opts&&opts.px)||0));g.connect(pan);pan.connect(buses.perc.g);} else g.connect(buses.perc.g);
    noise.start(t);noise.stop(t+duration+0.01);
    return true;
  }
  ORC.note = function (inst, midi, when, dur, vel, opts) {
    if (!ctx || !ready) return false;
    if (inst === 'battle_sword' || inst === 'battle_arrow') return battleFx(inst, +when, +vel || 0.4, opts);
    midi = Math.round(+midi); when = +when; dur = +dur; vel = +vel;
    if (!isFinite(midi) || midi < 14 || midi > 118) return false;
    if (!isFinite(when)) when = ctx.currentTime + 0.02;
    if (!isFinite(dur) || dur <= 0) dur = 0.30;
    if (!isFinite(vel) || vel <= 0) vel = 0.65;
    opts = opts || {}; opts.midi = midi;
    var ch = CHAR[inst] || CHAR.string_ensemble_1;
    var s = nearest(inst, midi);
    if (!s) return false;
    var t0 = Math.max(when, ctx.currentTime + 0.004);
    var f0 = 440 * Math.pow(2, (midi - 69) / 12);
    /* tok çalım: nota ömrü uzatılır (legato) + dinamik kazanç */
    var hold = Math.max(0.08, dur * (ch.len || 1.1));
    var amp = Math.min(1.5, vel) * ch.g * (opts.amp || 1) / (1 + Math.abs(opts.pz || 0) * 0.45);
    if (inst === 'taiko_drum' || inst === 'timpani' || inst === 'melodic_tom') amp *= masterConfig.drum;
    // Yüksek perdede artan parlaklık, nota başına ölçülü zayıflatılır.
    if (!ch.perc && midi > 84) amp *= Math.max(0.48, 1 - (midi - 84) * 0.023);
    if ((inst === 'glockenspiel' || inst === 'tubular_bells') && midi > 84) amp *= Math.max(0.48, 1 - (midi - 84) * 0.025);
    amp *= 1 - 0.33 * sharpRisk(inst, midi, vel);
    var ok = voice(inst, ch, s, t0, hold, amp, vel, opts, f0);
    /* alt oktav gövde (vurmalı): gerçek davul ağırlığı */
    if (ok && SUB[inst] && midi - 12 >= 20) {
      var o2 = {}; for (var kk in opts) o2[kk] = opts[kk];
      o2.midi = midi - 12; o2.cut = 0.55;
      voice(inst, ch, nearest(inst, midi - 12) || s, t0 + 0.002, hold, amp * SUB[inst], vel * 0.9, o2, f0 * 0.5);
    }
    if (ok) activeLog[inst] = ctx.currentTime;
    if (voices > peaks) peaks = voices;
    return ok;
  };

  ORC.stats = function () {
    return { voices: voices, peak: peaks, stolen: stolen, max: VOICE_MAX,
      ready: ready, loaded: loadedCount, total: totalInst };
  };
  ORC.setVolume = function (v) { if (master) paramSmooth(master.gain, Math.max(0, Math.min(0.70, Number(v) || 0)), 0.08); };
  ORC.resume = function () { if (ctx && ctx.state === 'suspended') return ctx.resume(); return Promise.resolve(); };
  ORC.fadeOut = function (sec) { if (!gateGain || !ctx) return; var p = gateGain.gain; p.cancelScheduledValues(ctx.currentTime); p.setValueAtTime(p.value, ctx.currentTime); p.linearRampToValueAtTime(0.0001, ctx.currentTime + (sec || 0.4)); };
  ORC.fadeIn = function (sec) { if (!gateGain || !ctx) return; var p = gateGain.gain; p.cancelScheduledValues(ctx.currentTime); p.setValueAtTime(p.value, ctx.currentTime); p.linearRampToValueAtTime(1, ctx.currentTime + (sec || 0.6)); };
  ORC.toneMaster = {
    configure: function (o) {
      o = o || {}; ['low','mid','high'].forEach(function (k) { if (isFinite(o[k])) masterConfig[k] = Math.max(-4, Math.min(4, +o[k])); });
      if (isFinite(o.drum)) masterConfig.drum = Math.max(0.35, Math.min(1.00, +o.drum));
      if (isFinite(o.vibrato)) masterConfig.vibrato = Math.max(0, Math.min(0.45, +o.vibrato));
      if (isFinite(o.haze)) masterConfig.haze = Math.max(0, Math.min(1, +o.haze));
      if (isFinite(o.softness)) masterConfig.softness = Math.max(0, Math.min(1, +o.softness));
      if (typeof o.dynamic === 'boolean') masterConfig.dynamic = o.dynamic;
      if (typeof o.economy === 'boolean') masterConfig.economy = o.economy;
      if (typeof o.battle === 'boolean') masterConfig.battle = o.battle;
      if (lowEQ) { paramSmooth(lowEQ.gain, masterConfig.low); paramSmooth(midEQ.gain, masterConfig.mid); paramSmooth(highEQ.gain, masterConfig.high); }
    },
    stats: function () { return { peak: masterPeak, rms: masterRms, autoGain: autoGain, reduction: finalLimiter ? finalLimiter.reduction : 0, treble: trebleReduction, voices: voices, max: VOICE_MAX, state: ctx ? ctx.state : 'unavailable' }; }
  };
  ORC.familyOf = function (inst) { return FAMILY[inst] || 'str'; };
  ORC.noteName = noteName;
  root.Orchestra = ORC;
})(typeof window !== 'undefined' ? window : this);
