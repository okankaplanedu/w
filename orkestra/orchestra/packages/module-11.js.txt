      push('cello', ch[0] - 12, t, bar * 1.02, inDyn + 0.18);
      /* 4. ve 6. ölçüde zil kabarması */
      if (b === 3 || b === 6) push('reverse_cymbal', 60, t, bar * 0.9, 0.5 + b * 0.05);
      if (b === 7 && DR) {
        var tp7 = DR.timpPitches(ch);
        for (var r = 0; r < 16; r++) push('timpani', r % 4 === 3 ? tp7[1] : tp7[0], t + r * beat * 0.25, 0.1, 0.35 + r * 0.045);
      }
    }

    /* ---------------- 2) TEMA · 16 ölçü · fff ----------------
       Korno + trompet ana tema, çello kontrpuan, viyola arpej */
    var b1 = b0 + NB_INTRO, NB_THEME = 16;
    var ti = 0;
    for (var tb = 0; tb < NB_THEME; tb++) {
      var t2 = barT(b1 + tb);
      var c2 = CH[prog[(4 + Math.floor(tb / 2)) % prog.length]];
      chordAt[b1 + tb] = c2;
      if (MD && invPlan.length) c2 = MD.invert(c2, invPlan[Math.floor(tb / 2) % invPlan.length]);
      /* tema: her 4 ölçüde bir 4 ölçülük tema bloğu, ardından 2 ölçü uzatma */
      var blockStart = (tb % 4 === 0);
      if (blockStart) ti = tb === 0 ? 0 : Math.floor(R() * Math.max(1, theme.length));
      var ti2 = infoOf(b1 + tb);
      var pos = 0;
      var mLen = theme.length;
      var guard = 0;
      while (pos < 4 && guard++ < 64) {                    // ölçü DAİMA dolar
        var note = theme[ti % mLen];
        var durB = Math.max(0.25, note[1]);
        if (pos + durB > 4) durB = 4 - pos;
        var vv = 0.95 * (ti2 && ti2.dyn ? ti2.dyn : 1);
        // Güçlü vuruşlarda değişen armoniye bağlan; ara notalar motif karakterini korur.
        var melodic = (Math.abs(pos - Math.round(pos)) < 0.001) ? nearestChordPitch(note[0], c2) : note[0];
        push('trumpet', melodic, t2 + pos * beat, Math.max(0.12, durB * beat * 0.95), vv);
        push('violin', melodic + 12, t2 + pos * beat, Math.max(0.12, durB * beat * 0.95), 0.72 * (ti2 ? ti2.dyn : 1));
        if (tb % 2 === 0) push('french_horn', melodic - 12, t2 + pos * beat, Math.max(0.15, durB * beat), 0.66);
        /* motif hücresi süslemesi (depo) */
        if (SP && durB >= 1 && ti % 3 === 0) {
          for (var mq = 0; mq < motifCell.length; mq++) {
            push('oboe', 69 + motifCell[mq], t2 + pos * beat + mq * beat * 0.25, beat * 0.24, 0.42);
          }
        }
        pos += durB;
        ti++;
      }
      /* korno fanfar akorları: 1 ve 3 — AKOR SIRASI her ölçüde rastgele permütasyon */
      [0, 2].forEach(function (o, oi) {
        c2.slice(1).forEach(function (pc, j) {
          var oi2 = ordAt(b1 + tb, oi * 2 + j);
          push('french_horn', pc - 12 + (oi2 % 2 === 1 ? 12 : 0),
            t2 + (o + oi2 * 0.18) * beat, beat * 1.6, 0.70);
        });
      });
      /* çello kontrpuan: inici çizgi */
      var ctr = [c2[0] - 12, c2[1] - 12, c2[2] - 12, c2[Math.floor(R() * 3)] - 12];
      for (var si = ctr.length - 1; si > 0; si--) { var sj = Math.floor(R() * (si + 1)); var tmp = ctr[si]; ctr[si] = ctr[sj]; ctr[sj] = tmp; }
      ctr.forEach(function (m, j) {
        push('cello', m, t2 + j * beat, beat * (0.72 + R() * 0.18), 0.48 + R() * 0.12);
      });
      /* viyola arpej (16'lık) */
      for (var a2 = 0; a2 < 16; a2++) {
        /* AKOR SIRASI rastgele: her eserde/ölçüde farklı permütasyon */
        var am = c2[ordAt(b1 + tb, a2)] + (ordAt(b1 + tb, a2 + 1) === 0 && a2 >= 8 ? 12 : 0);
        push('viola', am, t2 + a2 * beat * 0.25, 0.11, 0.35 + (a2 % 4 === 0 ? 0.2 : 0));
      }
      /* ÖFKE / GERİLİM: akorun yeden ve triton bölgesinden yaklaşım (ölçek içi) */
      if (ti2 && ti2.tens > 0.3) {
        for (var tq = 0; tq < 4; tq++) {
          var appr = c2[0] + (tq % 2 ? 11 : 6);
          push('trombone', appr, t2 + tq * beat + beat * 0.75, beat * 0.24, 0.55 * ti2.tens + 0.3);
        }
      }
      /* TREMOLO: korku/kuşatma dokusu */
      if (ti2 && ti2.trem > 0.05) {
        var tN = 4 + Math.round(ti2.trem * 16);
        for (var tr = 0; tr < tN; tr++) {
          push('tremolo_strings', c2[tr % 3] + 12, t2 + tr * (bar / tN), bar / tN * 0.9, 0.22 + ti2.trem * 0.5);
        }
      }
      /* kontrbas marcato */
      push('contrabass', c2[0] - 12, t2, beat * 0.6, 0.82);        // bir oktav yukarı = daha az çamur
      push('contrabass', c2[0] - 12, t2 + 2 * beat, beat * 0.6, 0.70);
      /* --- İKİ ORDU: her ölçüde sağ ve sol blok ayrı aileden, karşıt panla --- */
      if (FPLAN) {
        var wC = FPLAN.weigh(b1 + tb);
        var pw = powAt(t2);
        /* sağ blok: 1. ordu */
        push('brass_section', c2[0], t2, bar * 0.62, 0.30 * wC.merkez * palette.brass,
          FPLAN.sidePan(1));
        push('trumpet', c2[2] + 12, t2 + beat * 0.5, beat * 1.2, 0.26 * wC.suvarı * palette.brass,
          FPLAN.sidePan(1));
        /* sol blok: 2. ordu */
        push('french_horn', c2[1] - 12, t2, bar * 0.62, 0.30 * wC.solkanat * palette.brass,
          FPLAN.sidePan(0));
        push('trombone', c2[0] - 12, t2 + beat * 2, beat * 1.4, 0.26 * wC.topcu * palette.brass,
          FPLAN.sidePan(0));
        /* öncü kıta: sağ önde staccato */
        if (tb % 2 === 0) {
          for (var oc = 0; oc < 4; oc++) {
            push('violin', c2[oc % 3] + 12, t2 + oc * beat, beat * 0.42,
              0.30 * wC.oncu * palette.str, FPLAN.sidePan(1));
          }
        }
        /* --- GÜÇ DENGESİ OLAYI: aniden diğer kanattan yeni birlik girer --- */
        if (pw && Math.abs(pw.t - t2) < beat * 0.6) {
          var evT = t2 + (pw.t - t2);
          if (pw.type === 'karsi' || pw.type === 'dalga') {
            for (var q = 0; q < 6; q++) {
              push('trumpet', c2[q % 3] + (q > 2 ? 12 : 0), evT + q * beat * 0.5, beat * 0.9,
                pw.amp * palette.brass, pw.pan1);
              push('timpani', c2[0] - 12 + (q % 2 ? 7 : 0), evT + q * beat * 0.5, 0.12, pw.amp);
            }
          } else if (pw.type === 'kitalar') {
            push('brass_section', c2[0] - 12, evT, bar * 0.5, pw.amp * palette.brass, pw.pan0);
            push('brass_section', c2[0] - 5, evT, bar * 0.5, pw.amp * palette.brass, pw.pan1);
            push('taiko_drum', c2[0] - 24, evT, 0.14, pw.amp);
          } else if (pw.type === 'pan') {
            push('viola', c2[1], evT, bar * 0.4, pw.amp * palette.str, pw.pan0);
            push('viola', c2[2], evT + beat, bar * 0.4, pw.amp * palette.str, pw.pan1);
          } else if (pw.type === 'cephegeri') {
            push('oboe', c2[2] + 12, evT, bar * 0.9, pw.amp * 0.55, { px: 0.06, pz: -2.2 });
            push('choir_aahs', c2[0] + 12, evT, bar * 1.1, pw.amp * 0.40, { px: 0.05, pz: -2.6 });
          }
        }
      }
      /* --- YAYLI YATAK: her ölçüde tam akor (orkestra gövdesi duyulsun) --- */
      var susV = 0.52 * (ti2 ? ti2.dyn : 1);
      push('string_ensemble_1', c2[0], t2, bar * 1.02, susV);
      push('string_ensemble_1', c2[1], t2, bar * 1.02, susV * 0.88);
      push('string_ensemble_1', c2[2], t2, bar * 1.02, susV * 0.82);
      push('string_ensemble_2', c2[0] + 12, t2, bar * 1.02, susV * 0.60);
      push('string_ensemble_2', c2[2] + 12, t2, bar * 1.02, susV * 0.52);
      push('cello', c2[0] - 12, t2, bar * 1.02, 0.66 * (ti2 ? ti2.dyn : 1));
      push('bassoon', c2[0] - 12, t2, bar * 1.02, 0.40 * (ti2 ? ti2.dyn : 1));
      /* VURMALI: fazın ritmik hücresi, akora akortlu */
      if (DR) {
        DR.play({ push: push, barStart: t2, beat: beat, cell: ti2.perc, chord: c2,
          dyn: ti2.dyn, dens: ti2.dens, R: R0, seedBar: tb + 2 });
        /* karşı kıta vuruşları: uzaktan yaklaşan alay (3D) */
        if (Math.abs(ti2.b) > 0.45 && tb % 2 === 0) {
          DR.regiment({ push: push, barStart: t2 + beat * 2, beat: beat, R: R0,
            side: ti2.b > 0 ? -1 : 1, dist: Math.abs(ti2.b) * 0.9, dyn: 0.7 });
        }
      }
      /* --- KORO: yalnızca DORUK ölçülerinde (tb 7-8, 14-15) — sürekli yastık yok --- */
      var peakBar = (tb === 6 || tb === 7 || tb === 14 || tb === 15);
      if (arr.choir !== 'yok' && peakBar) {
        var cv = choirVoicing(c2, arr.choir === 'tutti' ? 'aahs' : arr.choir);
        cv = cv.slice();
        for (var cvj = cv.length - 1; cvj > 0; cvj--) {          // rastgele sıra
          var cvi = Math.floor(R0() * (cvj + 1)); var cvt = cv[cvj]; cv[cvj] = cv[cvi]; cv[cvi] = cvt;
        }
        cv.forEach(function (m3, j3) {
          push(arr.choir === 'oohs' ? 'voice_oohs' : 'choir_aahs', m3 + 12, t2, bar * 0.9,
            ((tb >= 14 ? 0.46 : 0.36) + (j3 === 0 ? 0.05 : 0)) * (palette.choir || 1));
        });
      }
      if (tb % 2 === 1) {
        var wl = windLine(c2, R0, 12);
        wl.forEach(function (m4, j4) {
          push(pick(R0, ['flute', 'oboe', 'clarinet']), m4, t2 + j4 * beat, beat * 0.9,
            0.50 * (palette.wood || 1));
        });
      }
      if (arr.id === 'katedral' || arr.id === 'konser') {
        for (var h0 = 0; h0 < 8; h0++) {
          push('orchestral_harp', c2[h0 % 3] + (h0 > 3 ? 12 : 0), t2 + h0 * beat * 0.5, beat * 0.45, 0.34);
        }
      }
      /* --- ÇAN RENGİ: palet çanı yükselttiyse tiz ışıltı, yoksa çok hafif --- */
      var bellW = palette.bell || 1;
      if (tb % 4 === 2) {
        var gn = c2[2] + 24;
        push('glockenspiel', gn, t2 + beat * 1.5, beat * 0.6, 0.22 * bellW);
        push('glockenspiel', c2[1] + 24, t2 + beat * 2.5, beat * 0.5, 0.18 * bellW);
        if (bellW > 1.15) {
          push('tubular_bells', c2[0] + 12, t2, bar * 0.8, 0.26 * bellW);
        }
      }
      /* --- ALTIN KESİT (φ = 0.618): enerji doruk noktası --- */
      if ((b1 + tb) === GOLDEN) {
        var gch = chordAt[b1 + tb] || c2;
        push('orchestra_hit', 57, t2, bar * 0.8, 1.15);
        push('timpani', DR ? DR.timpPitches(gch)[0] : gch[0] - 12, t2, 0.5, 1.1);
        push('taiko_drum', gch[0] - 24, t2, 0.3, 1.05);
        push('reverse_cymbal', 60, t2, bar * 0.9, 0.9);
        push('brass_section', gch[0], t2, bar * 0.95, 0.9);
        push('trumpet', gch[2] + 12, t2, bar * 0.9, 0.95);
        push('string_ensemble_1', gch[0], t2, bar * 1.0, 0.7);
        push('string_ensemble_1', gch[2], t2, bar * 1.0, 0.65);
        push('tubular_bells', gch[0] + 12, t2, bar * 1.2, 0.42);
      }
      if (tb === 7) push('orchestra_hit', 57, t2, 1.2, 0.95);
      if (tb === 15) {                                              /* doruk öncesi kabarma */
        for (var g0 = 0; g0 < 16; g0++) push('timpani', c2[0] - 12, t2 + g0 * beat * 0.25, 0.1, 0.4 + g0 * 0.04);
        push('reverse_cymbal', 60, t2, bar * 0.95, 0.85);
      }
    }

    /* ---------------- 3) KÖPRÜ · 12 ölçü · mf → ff ----------------
       Perküsyon solosu, tremolo yaylılar, pizzicato atışlar */
    var b2 = b1 + NB_THEME, NB_BRIDGE = 12;
    var b3 = b2;                                  /* köprü bloğu b3'te başlar */
    for (var bb = 0; bb < NB_BRIDGE; bb++) {
      var t3 = barT(b2 + bb);
      var c3 = CH[prog[(9 + Math.floor(bb / 2)) % prog.length]];
      chordAt[b3 + bb] = c3;
      var lvl = 0.45 + (bb / NB_BRIDGE) * 0.55;                     // kabaran gerilim
      /* tremolo yaylı yastık */
      push('tremolo_strings', c3[0], t3, bar * 0.99, 0.26 * lvl + 0.1);
      push('tremolo_strings', c3[2] + 12, t3, bar * 0.99, 0.22 * lvl + 0.08);
      /* VURMALI: kuşatma/karşı saldırı hücresi (akortlu, ızgarada) */
      var bi3 = infoOf(b2 + bb);
      if (DR) {
        DR.play({ push: push, barStart: t3, beat: beat, cell: bi3.perc, chord: c3,
          dyn: bi3.dyn * (0.85 + bb * 0.015), dens: bi3.dens, R: R0, seedBar: bb + 5,
          cymbal: (bb === 11) });
        if (bb % 3 === 0) {
          DR.regiment({ push: push, barStart: t3, beat: beat, R: R0,
            side: bb % 2 ? 1 : -1, dist: 0.8 * (1 - bb / 12), dyn: 0.6 + bb * 0.03 });
        }
      }
      push('contrabass', c3[0] - 12, t3, bar * 0.98, 0.55 * lvl + 0.25);
      push('bassoon', c3[0] - 12, t3, bar * 0.98, 0.42 * lvl + 0.2);
      /* pizzicato atışlar (senkop) */
      [0.5, 1.5, 2.75, 3.25].forEach(function (o2, j) {
        push('pizzicato_strings', c3[j % 3] + 12, t3 + o2 * beat, 0.14, 0.55 * lvl + 0.2);
      });
      /* koro kabarması (yalnızca köprünün son iki ölçüsü, oktav yukarıdan) */
      if (arr.choir !== 'yok' && bb >= 10) {
        choirVoicing(c3, 'aahs').forEach(function (m5, j5) {
          push(arr.choir === 'oohs' ? 'voice_oohs' : 'choir_aahs', m5 + 12, t3 + j5 * 0.02, bar * 0.85, 0.34);
        });
      }
      if (bb >= 8) {
        [0, 2].forEach(function (o3, j3) {
          c3.forEach(function (pc3) {
            push('brass_section', pc3 + (j3 === 1 ? 12 : 0), t3 + o3 * beat, beat * 0.9, 0.42 + (bb - 8) * 0.06);
          });
        });
        push('trombone', c3[0] - 12, t3 + 2 * beat, beat * 1.8, 0.55);
      }
      /* glockenspiel parıltı (köprünün sonunda) */
      if (bb === 9 || bb === 11) {                    // seyrek, yüksek, temiz parıltı
        [84, 91].forEach(function (mq, q2) {
          push('tubular_bells', mq + 12, t3 + q2 * 2 * beat, 1.1, 0.26);
        });
      }
      if (bb === 6) push('glockenspiel', 96, t3, 0.6, 0.22);
      if (bb === 11) {                                              /* köprü çıkışı: tam orkestra çarpması */
        push('orchestra_hit', 57, t3 + 3 * beat, 1.6, 1.1);
        push('reverse_cymbal', 60, t3, bar * 0.9, 0.9);
      }
    }

    /* ---------------- 4) FİNAL · 10 ölçü · fff ----------------
       Tam orkestra kilitlenme + La minör kadans (A – E – A) */
    var b4 = b2 + NB_BRIDGE, NB_FINAL = 10;   /* final bloğu b4'te başlar */
    for (var fb = 0; fb < NB_FINAL; fb++) {
      var t4 = barT(b4 + fb);
      var isCad = fb >= NB_FINAL - 2;
      var c4 = isCad ? (fb === NB_FINAL - 2 ? CH.E : CH.Am) : CH[prog[(12 + Math.floor(fb / 2)) % prog.length]];
      chordAt[b4 + fb] = c4;
      /* tutti tema (trompet + keman oktav, korno + çello destek) */
      var bi = 0;
      while (bi < theme.length) {
        var nt = theme[bi];
        if (nt[0] > 0 && bi % 4 === 0 && fb < NB_FINAL - 2) {
          push('trumpet', nt[0], t4 + (bi % 4) * beat, Math.max(0.12, nt[1] * beat), 1.05);
          push('violin', nt[0] + 12, t4 + (bi % 4) * beat, Math.max(0.12, nt[1] * beat), 0.8);
          push('french_horn', nt[0] - 12, t4 + (bi % 4) * beat, Math.max(0.12, nt[1] * beat), 0.7);
          push('cello', nt[0] - 12, t4 + (bi % 4) * beat, Math.max(0.12, nt[1] * beat), 0.65);
        }
        bi += 4;
      }
      /* tam orkestra akoru (1. vuruş) */
      c4.forEach(function (pc, j) {
        push('brass_section', pc, t4, beat * 1.2, 0.85);
        push('string_ensemble_1', pc + (j === 2 ? 12 : 0), t4, beat * 1.2, 0.6);
      });
      /* final korosu: yalnızca doruk ve kadans ölçülerinde, tiz yerleşim */
      if (arr.choir !== 'yok' && (fb >= 4)) {
        choirVoicing(c4, 'aahs').forEach(function (m6, j6) {
          push(arr.choir === 'oohs' ? 'voice_oohs' : 'choir_aahs', m6 + 12, t4, beat * 1.2, 0.40 + j6 * 0.03);
        });
      }
      /* karşıtlama: aranjmanın çalgısı alt çizgiyi dolaşır */
      if (fb < NB_FINAL - 2) {
        var ctr2 = [c4[0] - 12, c4[1] - 12, c4[2] - 12, c4[1] - 12];
        ctr2.forEach(function (m7, j7) {
          push(arr.ctr, m7, t4 + j7 * beat, beat * 0.92, 0.6);
        });
      }
      /* --- FİNAL FULL ORKESTRA: her ölçüde tüm aileler birlikte --- */
      var finv = 0.55 + (fb / NB_FINAL) * 0.45;
      push('violin', c4[2] + 12, t4, bar * 0.98, 0.58 * finv * palette.str);
      push('violin', c4[0] + 12, t4 + beat * 0.5, bar * 0.9, 0.48 * finv * palette.str);
      push('viola', c4[1], t4, bar * 0.98, 0.46 * finv * palette.str);
      push('viola', c4[2], t4 + beat * 2, bar * 0.5, 0.42 * finv * palette.str);
      push('string_ensemble_2', c4[1] + 12, t4, bar * 1.02, 0.50 * finv * palette.str);
      push('string_ensemble_2', c4[2] + 12, t4, bar * 1.02, 0.44 * finv * palette.str);
      push('cello', c4[0] - 12, t4, bar * 1.02, 0.60 * finv * palette.str);
      push('french_horn', c4[1], t4, bar * 0.95, 0.56 * finv * palette.brass);
      push('trombone', c4[0] - 12, t4, bar * 0.95, 0.56 * finv * palette.brass);
      push('french_horn', c4[2] + 12, t4, bar * 0.95, 0.48 * finv * palette.brass);
      push('timpani', DR ? DR.timpPitches(c4)[0] : c4[0] - 12, t4, 0.35, 0.80 * finv);
      push('timpani', DR ? DR.timpPitches(c4)[1] : c4[0] - 5, t4 + beat * 2, 0.30, 0.66 * finv);
      push('tubular_bells', c4[0] + 12, t4, bar * 1.3, 0.26 * finv * palette.bell);
      push('glockenspiel', c4[2] + 24, t4 + beat, bar * 0.8, 0.22 * finv * palette.bell);
      push('flute', c4[2] + 24, t4 + beat * 0.5, bar * 0.85, 0.30 * finv * palette.wood);
      push('oboe', c4[1] + 12, t4, bar * 0.9, 0.30 * finv * palette.wood);
      push('clarinet', c4[0] + 12, t4 + beat, bar * 0.85, 0.28 * finv * palette.wood);
      push('bassoon', c4[0] - 12, t4, bar * 0.95, 0.32 * finv * palette.wood);
      push('orchestral_harp', c4[2], t4 + beat * 1.5, bar * 0.6, 0.32 * finv * palette.wood);
      if (arr.choir !== 'yok' && fb >= 2) {
        choirVoicing(c4, arr.choir === 'tutti' ? 'aahs' : arr.choir).forEach(function (mm9) {
          push(arr.choir === 'oohs' ? 'voice_oohs' : 'choir_aahs', mm9, t4, bar * 1.02,
            0.30 * finv * palette.choir);
        });
      }
      push('tuba', c4[0] - 24, t4, beat * 0.9, 0.85);
      push('contrabass', c4[0] - 24, t4, beat * 1.4, 0.9);
      var bi4 = infoOf(b3 + fb);
      if (DR) {
        DR.play({ push: push, barStart: t4, beat: beat, cell: bi4.perc, chord: c4,
          dyn: bi4.dyn, dens: bi4.dens, R: R0, seedBar: fb + 9, cymbal: false });
      }
      if (fb === 2 || fb === 5) push('orchestra_hit', 57, t4, 1.4, 1.0);
      /* kadans: A – E – A */
      if (isCad) {
        var cadNotes = fb === NB_FINAL - 2 ? [52, 56, 59] : [57, 60, 64, 69];
        cadNotes.forEach(function (m2, jj) {
          push('brass_section', m2, t4 + (fb === NB_FINAL - 2 ? 0 : 0), bar * 0.9, 1.05);
          push('violin', m2 + 12, t4, bar * 0.9, 0.75);
          push('trumpet', m2 + (jj === 0 ? 12 : 0), t4, bar * 0.85, 1.0);
          push('cello', m2 - 12, t4, bar * 0.9, 0.7);
        });
        if (DR) { var tpc = DR.timpPitches(c4); push('timpani', tpc[0], t4, bar * 0.85, 1.05); }
        /* --- FİNAL FULL ORKESTRA: son ölçülerde her aile aynı anda --- */
        if (fb >= NB_FINAL - 4) {
          var fv = (fb - (NB_FINAL - 4) + 1) / 4;

          push('string_ensemble_2', c4[1] + 12, t4, bar * 1.1, (0.44 + fv * 0.22) * palette.str);
          push('violin', c4[2] + 12, t4, bar * 1.0, (0.50 + fv * 0.25) * palette.str);
          push('viola', c4[1], t4, bar * 1.0, (0.42 + fv * 0.20) * palette.str);
          push('cello', c4[0] - 12, t4, bar * 1.1, (0.50 + fv * 0.22) * palette.str);
          push('contrabass', c4[0] - 24, t4, bar * 1.1, (0.52 + fv * 0.20) * palette.str);
          push('french_horn', c4[0], t4, bar * 1.0, (0.55 + fv * 0.25) * palette.brass);
          push('french_horn', c4[2], t4, bar * 1.0, (0.48 + fv * 0.22) * palette.brass);
          push('trumpet', c4[2] + 12, t4, bar * 1.0, (0.55 + fv * 0.28) * palette.brass);
          push('trumpet', c4[0] + 12, t4, bar * 1.0, (0.50 + fv * 0.24) * palette.brass);
          push('trombone', c4[0] - 12, t4, bar * 1.0, (0.58 + fv * 0.24) * palette.brass);
          push('tuba', c4[0] - 24, t4, bar * 1.0, (0.60 + fv * 0.22) * palette.brass);
          push('brass_section', c4[1], t4, bar * 1.0, (0.52 + fv * 0.24) * palette.brass);
          push('timpani', DR ? DR.timpPitches(c4)[0] : c4[0] - 12, t4, 0.4, (0.85 + fv * 0.3));
          push('taiko_drum', c4[0] - 24, t4, 0.3, (0.80 + fv * 0.3));
          push('tubular_bells', c4[0] + 12, t4, bar * 1.2, (0.30 + fv * 0.25) * palette.bell);
          push('glockenspiel', c4[2] + 24, t4, bar * 1.0, (0.28 + fv * 0.24) * palette.bell);
          push('orchestral_harp', c4[0] + 12, t4, bar * 1.0, (0.34 + fv * 0.20) * palette.wood);
          push('flute', c4[2] + 24, t4, bar * 0.95, (0.30 + fv * 0.22) * palette.wood);
          push('oboe', c4[1] + 12, t4, bar * 0.95, (0.30 + fv * 0.20) * palette.wood);
          push('clarinet', c4[0] + 12, t4, bar * 0.95, (0.28 + fv * 0.20) * palette.wood);
          push('bassoon', c4[0] - 12, t4, bar * 0.95, (0.34 + fv * 0.18) * palette.wood);
          if (arr.choir !== 'yok') {
            choirVoicing(c4, arr.choir === 'tutti' ? 'aahs' : arr.choir).forEach(function (mm8) {
              push(arr.choir === 'oohs' ? 'voice_oohs' : 'choir_aahs', mm8, t4, bar * 1.05,
                (0.34 + fv * 0.24) * palette.choir);
            });
          }
          push('orchestra_hit', 57, t4, bar * 0.8, (0.70 + fv * 0.3));
          push('reverse_cymbal', 60, t4, bar * 0.9, (0.42 + fv * 0.22));
        }
        if (fb >= NB_FINAL - 3) {
          push('tubular_bells', c4[0] + 12, t4, bar * 1.2, 0.30 * (palette.bell || 1));
          push('glockenspiel', c4[2] + 24, t4 + beat, beat * 0.8, 0.24 * (palette.bell || 1));
          choirVoicing(c4, 'oohs').forEach(function (m7) {
            push('voice_oohs', m7 + 12, t4, bar * 1.05, 0.26 * (palette.choir || 1));
          });
        }
        push('orchestra_hit', 57, t4, bar * 0.95, 1.15);
        push('taiko_drum', 40, t4, 0.16, 1.1);
        if (fb === NB_FINAL - 1) {
          /* son akor: uzun La minör + zil */
          push('tubular_bells', 69, t4, 2.4, 0.5);
          push('reverse_cymbal', 60, t4, 2.0, 0.8);
          push('string_ensemble_1', 57, t4, bar * 1.6, 0.55);
        }
      }
    }

    /* ---- HORN/TUBA GARANTİSİ: kaos/jest sonrası eksik kalan ölçüleri doldur ---- */
    (function () {
      var hb2 = {};
      ev.forEach(function (e) {
        if (e.inst === 'french_horn' || e.inst === 'tuba') {
          var bb = Math.floor(e.t / bar);
          if (bb >= 0 && bb < 46) hb2[bb] = 1;
        }
      });
      for (var b2 = 0; b2 < 46; b2++) {
        if (hb2[b2]) continue;
        /* son kapı: bu ölçüde horn/tuba yok → yastık yaz */
        var cb2b = chordAt[b2] || CH.Am, tb2 = b2 * bar;
        push('french_horn', cb2b[0], tb2, bar * 1.02, 0.42);
        push('french_horn', cb2b[2], tb2, bar * 1.02, 0.34);
        push('tuba', cb2b[0] - 24, tb2, bar * 1.02, 0.56);
      }
    })();

    /* ---- 3D KITA YERLEŞİMİ: her olay kendi kıtasının uzay noktasından çalar,
       cephe kaydıkça kıtalar yer değiştirir (savaş alanı algısı) ---- */
    /* jestlerden SONRA çalışır: heliz panı uygulanmış olayların konumu KORUNUR */
    if (BT && bPlan) {
      ev.forEach(function (e) {
        if (e.px != null) return;                       // alay/heliz konumu korunur
        var bi = Math.min(bPlan.map.length - 1, Math.floor((e.t / (bar * 46)) * bPlan.map.length));
        var mm = bPlan.map[Math.max(0, bi)] || { b: 0 };
        var fam = BT.familyOf(e.inst);
        var pos = bPlan.regPos(fam, mm.b);
        e.px = +pos.x.toFixed(3);
        e.pz = +pos.z.toFixed(3);
        e.bar = bi;
      });
    }

    /* ---- DETONE SÜZGECİ: her perde La minör (harmonik) içinde kalır ---- */
    var SCALE = [9, 11, 0, 2, 4, 5, 7, 8];           // A B C D E F G G#
    var fixed = 0, dropped = 0;
    ev = ev.filter(function (e) {
      var pc = ((e.midi % 12) + 12) % 12;
      if (SCALE.indexOf(pc) >= 0) return true;
      var best = null, bd = 99;
      SCALE.forEach(function (sc2) {
        var d = Math.abs(((sc2 - pc + 12) % 12 + 12) % 12);
        if (d > 6) d = 12 - d;
        if (d < bd) { bd = d; best = sc2; }
      });
      var delta = ((best - pc + 12) % 12);
      if (delta > 6) delta -= 12;
      var nm = e.midi + delta;
      if (nm < 24 || nm > 100) { dropped++; return false; }
      e.midi = nm; fixed++;
      return true;
    });

    /* zaman sırası */
    /* ---- HORN + TUBA: sürekli nefesli gövde (her ölçüde) ---- */
    (function () {
      for (var hb = 0; hb < 46; hb++) {
        var th = hb * bar, chh = chordAt[hb] || CH.Am;
        var hi2 = infoOf(hb), hc = CEASE_BARS[hb] ? 0.5 : 1;
        var hv = Math.max(0.55, Math.min(1.25, hi2.dyn)) * hc;
        /* korno üçlüsü: akorun tamamı, uzun ve tok */
        push('french_horn', chh[0], th, bar * 1.02, 0.46 * hv);
        push('french_horn', chh[1], th, bar * 1.02, 0.40 * hv);
        push('french_horn', chh[2], th, bar * 1.02, 0.36 * hv);
        /* korno oktavı: tiz parlama */
        push('french_horn', chh[0] + 12, th, bar * 0.98, 0.26 * hv);
        /* tuba: sürekli bas, kontrbası ikiye katlar */
        push('tuba', chh[0] - 24, th, bar * 1.02, 0.62 * hv);
        push('tuba', chh[0] - 24, th + beat * 2, beat * 1.6, 0.50 * hv);
        push('contrabass', chh[0] - 24, th, bar * 1.02, 0.52 * hv);
        /* 1. ve 3. vuruşta korno-tuba tutti vurgusu (inançlı metronom) */
        push('brass_section', chh[0], th, beat * 0.85, 0.50 * hv);
        push('brass_section', chh[2], th + beat * 1.02, beat * 0.85, 0.42 * hv);
        if (hb % 4 === 0) {
          /* her 4 ölçüde korno fanfarı (savaş çağrısı) */
          push('trumpet', chh[2] + 12, th, beat * 0.7, 0.62 * hv);
          push('french_horn', chh[1] + 12, th + beat * 0.5, beat * 0.7, 0.58 * hv);
          push('trombone', chh[0], th + beat * 0.5, beat * 0.9, 0.55 * hv);
        }
        if (hb === GOLDEN) {
          push('tuba', chh[0] - 36, th, bar * 1.1, 0.70);
          push('french_horn', chh[2] + 12, th, bar * 1.05, 0.62);
        }
      }
    })();

    /* ---- CEPHE GERİSİ: endişe duaları, uzakta, her bestede rastgele ölçülerde ---- */
    if (FPLAN) {
      var worryB = FPLAN.worryBars, prayB = FPLAN.prayerBars;
      for (var wb = 0; wb < worryB.length; wb++) {
        var bb2 = Math.max(0, Math.min(45, worryB[wb]));
        var cb2 = chordAt[bb2] || CH.Am, tw = bb2 * bar;
        var inf2 = infoOf(bb2);
        /* endişe: obua + klarinet ikili, uzakta, hafif kromatik */
        push('oboe', cb2[2] + 12, tw, bar * 0.9, 0.34 * inf2.dyn, { px: 0.12, pz: -2.4 });
        push('clarinet', cb2[1] - 12, tw + beat * 0.5, bar * 0.8, 0.28 * inf2.dyn, { px: -0.08, pz: -2.6 });
        push('tremolo_strings', cb2[0] + 12, tw, bar * 0.95, 0.22 * inf2.dyn, { px: -0.18, pz: -2.0 });
        /* soru-cevap: obua bir yarım ton ile "kaygı" bırakır (ölçek içi) */
        /* kaygı: ölçek içi altılı (detone yasak) */
        var worryM = CHS ? CHS.snapScale(cb2[2] + 8) : cb2[2] + 8;
        push('oboe', worryM, tw + bar * 0.62, beat * 1.2, 0.26 * inf2.dyn, { px: 0.10, pz: -2.4 });
      }
      for (var pb = 0; pb < prayB.length; pb++) {
        var bb3 = Math.max(0, Math.min(45, prayB[pb]));
        var cb3 = chordAt[bb3] || CH.Am, tp2 = bb3 * bar;
        /* dua: koro uzaktan, iki kıta hâlinde alçak seste */
        choirVoicing(cb3, 'aahs').forEach(function (m6, j6) {
          push('choir_aahs', m6, tp2, bar * 1.15, 0.30, { px: -0.12 + j6 * 0.06, pz: -3.0 });
        });
        push('voice_oohs', cb3[2] + 12, tp2 + beat, bar * 0.9, 0.24, { px: 0.14, pz: -3.2 });
        push('bassoon', cb3[0] - 12, tp2, bar * 1.1, 0.26, { px: 0.0, pz: -2.0 });
      }
    }

    /* ---- BÜYÜK DERİ DAVULLARI: inançlı metronom, 46 ölçünün TAMAMINDA ----
       taiko (ana darbe) + timpani (akortlu) + melodic_tom (cevap) + woodblock (adım)
       sus/ateşkes bölgelerinde de çalar; yalnızca seviyesi düşer.                     */
    var drumBars = 0, drumHits = 0;
    (function () {
      for (var db = 0; db < 46; db++) {
        var td = db * bar;
        var cd = chordAt[db] || CH.Am;
        var inf = infoOf(db);
        var ceasing = !!CEASE_BARS[db];
        var lvl = (ceasing ? 0.55 : 1) * Math.max(0.62, Math.min(1.25, inf.dyn));
        var tp = DR ? DR.timpPitches(cd) : [cd[0] - 12, cd[0] - 5];
        /* ana darbe: taiko + timpani birlikte (göğüs vuruşu) */
        push('taiko_drum', cd[0] - 24, td, 0.16, 0.95 * lvl);
        push('timpani', tp[0], td, 0.22, 0.90 * lvl);
        /* 3. vuruş: timpani beşli (metronomun ikinci ayağı) */
        push('timpani', tp[1], td + beat * 2, 0.20, 0.78 * lvl);
        push('taiko_drum', cd[0] - 17, td + beat * 2, 0.14, 0.72 * lvl);
        /* 2 ve 4: melodic_tom cevap (hafif senkop) */
        push('melodic_tom', cd[0] - 12, td + beat * 1, 0.12, 0.55 * lvl);
        push('melodic_tom', cd[0] - 12, td + beat * 3, 0.12, 0.50 * lvl);
        /* BÜYÜK DERİ DAVULU: derin tom darbesi (göğüste hissedilir) */
        push('melodic_tom', cd[0] - 24, td, 0.18, 0.80 * lvl);
        push('melodic_tom', cd[0] - 24, td + beat * 2.5, 0.14, 0.62 * lvl);
        /* timpani çift darbe: metronomun inançlı nabzı */
        push('timpani', tp[0], td + beat * 1, 0.12, 0.52 * lvl);
        push('timpani', tp[1], td + beat * 3, 0.12, 0.48 * lvl);
        /* 16'lık adım: woodblock yürüyüşü (metronom nabzı) */
        for (var wq = 0; wq < 4; wq++) {
          push('woodblock', 72, td + wq * beat + beat * 0.5, 0.07, 0.24 * lvl);
        }
        /* faz doruklarında ek vuruş (kuşatma/karşı/zafer) */
        if (!ceasing && (inf.phase === 'zafer' || inf.phase === 'karsi' || inf.phase === 'kusatma')) {
          push('taiko_drum', cd[0] - 24, td + beat * 3.5, 0.12, 0.80 * lvl);
          push('timpani', tp[0], td + beat * 1.5, 0.16, 0.62 * lvl);
        }
        drumBars++; drumHits += 12;
      }
    })();

    /* ---- BOŞ ÖLÇÜ KAPISI: her ölçüde en az 3 olay (yastık + bas + vurgu) ---- */
    var TOTAL_BARS = 46;
    var barCount = {}, filled = 0;
    ev.forEach(function (e) { var bi = Math.floor(e.t / bar); barCount[bi] = (barCount[bi] || 0) + 1; });
    for (var fb2 = 0; fb2 < TOTAL_BARS; fb2++) {
      if (!barCount[fb2]) {
        var chf = chordAt[fb2] || CH.Am, tf = fb2 * bar;
        var inf = infoOf(fb2);
        push('string_ensemble_2', chf[0], tf, bar * 0.98, 0.30 * inf.dyn);
        push('string_ensemble_2', chf[2] + 12, tf, bar * 0.98, 0.24 * inf.dyn);
        push('cello', chf[0] - 12, tf, bar * 0.9, 0.34 * inf.dyn);
        push('contrabass', chf[0] - 12, tf + 2 * beat, beat * 0.6, 0.36);
        if (DR) DR.play({ push: push, barStart: tf, beat: beat, cell: inf.perc, chord: chf,
          dyn: inf.dyn, dens: inf.dens, R: R0, seedBar: fb2 });
        filled++;
      }
    }

    /* ---- VURMALI AKORT KAPISI: her timpani vuruşu ölçünün akoruna çekilir ---- */
    var PERC_SET = { timpani: 1, taiko_drum: 1, melodic_tom: 1, woodblock: 1, reverse_cymbal: 1 };
    var timpFixed = 0, timpSeen = 0;
    ev.forEach(function (e) {
      if (e.inst !== 'timpani') return;
      timpSeen++;
      var bi = Math.max(0, Math.floor(e.t / bar));
      var ch = chordAt[Math.min(bi, 45)] || chordAt[bi] || CH.Am;
      var tp = DR ? DR.timpPitches(ch) : [ch[0], ch[0] + 7];
      var okp = tp.some(function (q) { return ((q % 12) + 12) % 12 === ((e.midi % 12) + 12) % 12; });
      if (!okp) {
        /* en yakın akortlu tele çek (detone sıfır) */
        var tgt = tp[0], best = 1e9;
        tp.forEach(function (q) {
          for (var oct = -2; oct <= 2; oct++) {
            var cand = q + oct * 12;
            var dd = Math.abs(cand - e.midi);
            if (dd < best) { best = dd; tgt = cand; }
          }
        });
        e.midi = tgt; timpFixed++;
      }
    });

    /* ---- SUSLAR / ATEŞKESLER: orkestra nefes alır, düzen bozulmadan boşluk ---- */
    var ceaseInfo = null;
    if (CFP && CFP.length) {
      var removed = 0, kept = 0, detailC = [];
      CFP.forEach(function (w) {
        var t0 = w.start * bar, t1 = (w.start + w.len) * bar;
        var chW = chordAt[w.start] || CH.Am;
        var out = [];
        for (var i = 0; i < ev.length; i++) {
          var e = ev[i];
          if (e.t < t0 || e.t >= t1) { out.push(e); continue; }
          if (w.kind === 'fisilti') {                      // fısıltı: yalnız koro + tremolo kalır
            var kk = (e.inst === 'voice_oohs' || e.inst === 'choir_aahs' ||
              e.inst === 'tremolo_strings' || e.inst === 'timpani');
            if (kk) { e.v = e.v * 0.38; out.push(e); kept++; } else removed++;
            continue;
          }
          /* tam sus ve nefes: her şey silinir — AMA DAVUL ASLA SUSMAZ */
          if (e.t - t0 < (t1 - t0) * w.cutoff) {
            if (PERC_KEEP[e.inst]) { e.v = Math.max(0.34, e.v * 0.78); out.push(e); kept++; continue; }
            /* korno ve tuba da susmaz: alçak, uzun nefes (dua) */
            if (e.inst === 'french_horn' || e.inst === 'tuba' || e.inst === 'contrabass') {
              if (e.d > beat * 0.7) { e.v = Math.max(0.16, e.v * 0.42); out.push(e); kept++; continue; }
            }
            removed++; continue;
          }
          /* süren notalar susun başında KESİLİR → sus gerçekten duyulur */
          if (e.t < t0 && e.t + e.d > t0) { e.d = Math.max(0.05, +(t0 - e.t).toFixed(3)); }
          out.push(e); kept++;
        }
        ev = out;
        /* --- ATEŞKES BOŞLUĞU: solo çalgı uzakta, tek başına --- */
        if (w.kind !== 'fisilti') {
          var tS = t0 + (t1 - t0) * w.cutoff;
          var soloM = CHS ? CHS.snapScale(chW[2] + 12) : chW[2] + 12;
          push(w.solo, soloM, tS + beat * 0.5, beat * 1.6, 0.42, { px: 0.12, pz: -2.2 });
          push(w.solo, CHS ? CHS.snapScale(chW[1] + 12) : chW[1] + 12, tS + beat * 2.3, beat * 1.5, 0.36,
            { px: 0.02, pz: -2.6 });
          /* sustuktan sonra yeniden doğuş: tam ızgarada tutti giriş */
          push('timpani', chW[0] - 12, t1, 0.3, 0.9);
          push('brass_section', chW[0], t1, beat * 1.4, 0.72);
          push('string_ensemble_1', chW[0], t1, bar * 0.9, 0.62);
          push('string_ensemble_1', chW[2], t1, bar * 0.9, 0.52);
        } else {
        }
        detailC.push({ start: w.start, len: w.len, kind: w.kind, solo: w.solo, cut: +w.cutoff.toFixed(2) });
      });
      ceaseInfo = { windows: detailC.length, removed: removed, kept: kept, detail: detailC };
      /* sus bölgeleri: kaos/jest/güç olayları buralara GİREMEZ (sessizlik korunur) */
      CFP.forEach(function (w) {
        if (w.kind === 'fisilti') return;                 // fısıltıda doku kalır
        for (var q = w.start; q <= w.start + w.len; q++) CEASE_BARS[q] = 1;
      });
    }

    /* ---- sus bölgelerini kaos pencerelerinden çıkar ---- */
    if (KPLAN && KPLAN.windows) {
      KPLAN.windows = KPLAN.windows.filter(function (w) {
        for (var q = w.start; q < w.start + w.len + 1; q++) if (CEASE_BARS[q]) return false;
        return true;
      });
    }

    /* ---- KAOS UYGULA: faz kayması / polimetri / küme / kanon / kayma + çözülme ---- */
    var chaosInfo = null;
    if (CHS && KPLAN) {
      chaosInfo = CHS.apply({
        events: ev, plan: KPLAN, bar: bar, beat: beat, tempo: tempo, R: R0,
        push: push, chordAt: chordAt, palette: palette,
        dyn: (bPlan && bPlan.map[0] ? bPlan.map[0].dyn : 1) || 1,
        chord: CH[prog[0]]
      });
    }

    /* ---- JESTLER: heliz pan, tümsek/çukur, çığlık, galeyan, yalvarma ---- */
    var jestInfo = null;
    if (root.Jest) {
      var JPLAN = root.Jest.plan({ bars: 46, seed: seed });
      /* sus bölgelerini jestlerden arındır */
      ['rises', 'falls', 'screams', 'rallies', 'begs'].forEach(function (k) {
        JPLAN[k] = JPLAN[k].filter(function (w) {
          for (var q = w.start; q < w.start + (w.len || 1) + 1; q++) if (CEASE_BARS[q]) return false;
          return true;
        });
      });
      jestInfo = root.Jest.apply({
        events: ev, plan: JPLAN, bar: bar, beat: beat, tempo: tempo, R: R0,
        push: push, chordAt: chordAt, palette: palette,
        dyn: (bPlan && bPlan.map[0] ? bPlan.map[0].dyn : 1) || 1,
        chord: CH[PROG_BUILT[0]]
      });
    }

    if (bPlan) { HIST.push(bPlan.order.slice()); if (HIST.length > 24) HIST.shift();
      root.__prevPlan = { order: bPlan.order.slice(), map: bPlan.map, mirror: true }; }
    ev.sort(function (a, b) { return a.t - b.t; });

    /* ---- DURUM TEMPOLARI: her bölüm, o bölümdeki cephe durumuna göre tempo alır ---- */
    var SEC_TEMPO = {
      esit: 0, ilerleme: 3, baskin: 6, gerileme: -4,
      kusatma: -6, karsi: 8, zafer: 5
    };
    function phaseOfBars(a, b) {                 // a..b ölçü aralığının baskın fazı
      if (!bPlan) return 'esit';
      var cnt = {};
      for (var i = a; i < b && i < bPlan.map.length; i++) {
        var ph = bPlan.map[i].phase;
        cnt[ph] = (cnt[ph] || 0) + 1;
      }
