(function (root) {
  'use strict';
  var A = root.Orchestra, S = root.Score;
  var state = {
    playing: false, score: null, nextIdx: 0, startAt: 0, loop: 0,
    loopCount: 0, timer: null, lookAhead: 0.9, secIdx: 0, notesPlayed: 0,
    lastCycleAt: 0, activeInst: {}
  };

  function $(s) { return document.querySelector(s); }
  function fmt(sec) {
    if (!isFinite(sec) || sec < 0) sec = 0;
    var m = Math.floor(sec / 60), s2 = Math.floor(sec % 60);
    return m + ':' + (s2 < 10 ? '0' : '') + s2;
  }

  var lastArr = -1, lastArrId = '';
  function newScore(first) {
    /* her yeni şarkı FARKLI aranjman: en son kullanılanı atla */
    var n = 6;
    var pickIdx = Math.floor(Math.random() * n);
    if (pickIdx === lastArr) pickIdx = (pickIdx + 1 + Math.floor(Math.random() * (n - 1))) % n;
    lastArr = pickIdx;
    var seed0 = (Date.now() ^ (state.loopCount * 2654435761)) >>> 0;
    /* ---- KISITLI RASTGELELİK: tabu geçmişine göre en özgün adayı seç ---- */
    var result = null;
    if (root.Model) {
      result = root.Model.generate(function (i) {
        var s2 = S.build({ seed: (seed0 + i * 7919) >>> 0, arr: (pickIdx + i) % 6 });
        s2.arrRepeat = (s2.arrId === lastArrId);
        return s2;
      }, { threshold: 0.62, attempts: 10 });
      state.score = result.song;
      root.__prevPlan = state.score.battle ? { order: state.score.battle.order, map: state.score.battle.map, mirror: true } : null;
      lastArrId = state.score.arrId;
      state.model = { score: +result.score.toFixed(3), history: result.history, mutated: !!result.mutated, mutP: result.song.mutP };
    } else {
      state.score = S.build({ seed: seed0, arr: pickIdx });
    }
    /* ---- SAYAÇ: imza ile birebir tekrarı engelle, benzersizliği işaretle ---- */
    if (root.Sayac) {
      var sy = root.Sayac;
      var sig = sy.signature(state.score);
      var dup = sy.seen(sig);
      state.score.benzersiz = !dup;
      var sg = state.score.sig || [];
      state.benz = null;
      try {
        var hist = JSON.parse(root.localStorage.getItem('oksl_sig') || '[]');
        var maxS = 0;
        hist.forEach(function (h) { maxS = Math.max(maxS, sy.similarity(sg, h)); });
        hist.push(sg); if (hist.length > 40) hist.shift();
        root.localStorage.setItem('oksl_sig', JSON.stringify(hist));
        state.benz = { max: maxS, n: hist.length };
      } catch (e) {}
      sy.push(sig);
      state.sayac = sy.report();
    }
    state.battle = state.score.battle;
    state.forces = state.score.forces;
    /* sus bölgeleri: çalma sırasında orkestra susar (gerçek ateşkes) */
    state.ceaseBars = {};
    if (state.score.ceasefire && state.score.ceasefire.detail) {
      state.score.ceasefire.detail.forEach(function (d) {
        for (var q = d.start; q <= d.start + d.len; q++) state.ceaseBars[q] = 1;
      });
    }
    paintForces();
    paintKimlik();
    state.nextIdx = 0;
    state.startAt = A.now() + 0.35;
    state.cycleStart = state.startAt;
    state.secIdx = 0;
    if (!first) state.loopCount++;
    var sc = state.score;
    if ($('#scInfo')) $('#scInfo').textContent =
      sc.key + ' · ' + sc.tempo + ' BPM · ' + sc.meter + ' · ' + fmt(sc.totalSec) +
      ' · ' + sc.events.length + ' nota · döngü #' + (state.loopCount + 1);
    if ($('#barInfo')) $('#barInfo').textContent = '0 / ' + sc.duration + ' ölçü';
    if ($('#arrPill')) $('#arrPill').textContent = sc.arrName;
    if ($('#bpmPill')) $('#bpmPill').textContent = sc.tempo;
    if ($('#arrInfo')) $('#arrInfo').textContent = sc.arrName + ' · koro ' + sc.choirMode +
      ' · ' + sc.events.length + ' olay · mutasyon p=' + (sc.mutP || 0).toFixed(2);
    if ($('#chaosInfo') && sc.chaos) {
      var cc = sc.chaos;
      $('#chaosInfo').textContent = 'kaos ' + cc.windows + ' faz · ' + cc.moved + ' kayma · ' +
        cc.maxShiftBeats.toFixed(2) + ' vuruş';
    }
    /* ---- BENZERSİZLİK SAYACI ---- */
    if ($('#uniqInfo')) {
      var sy2 = root.Sayac;
      var u = sy2 ? sy2.history() : 0;
      var tot2 = sy2 ? sy2.total() : 0;
      if ($('#m3Info')) {
        $('#m3Info').textContent = 'markov-3 ' + (state.score.m3 && state.score.m3.used ? 'aktif' : 'yok') +
          ' · cümle ' + (state.score.m3 ? state.score.m3.phraseLen : 0) + ' nota · φ altın kesit ölçü ' +
          (state.score.goldenBar || '—') + ' · φ bölümleme ' + (state.score.fibSplit || []).join('·');
      }
      $('#uniqInfo').textContent = 'benzersizlik ' + (state.score.benzersiz === false ? 'TEKRAR!' : 'YENİ') +
        (state.benz ? ' · en yakın benzerlik %' + (state.benz.max * 100).toFixed(0) : '') +
        ' · arşiv ' + u + ' · üretim ' + tot2;
      $('#uniqInfo').style.color = state.score.benzersiz === false ? '#e0607a' : '#7fd8c8';
    }
    if ($('#modelInfo')) {
      var mm = state.model || {};
      $('#modelInfo').textContent = 'tabu benzerlik ' + (mm.score != null ? (mm.score * 100).toFixed(0) + '%' : '—') +
        ' · geçmiş ' + (root.Model ? root.Model.historySize() : 0) + ' şarkı' + (mm.mutated ? ' · mutasyon uygulandı' : '');
    }
    if ($('#loopInfo')) $('#loopInfo').textContent = String(state.loopCount + 1);
  }

  /* ---- İKİ ORDU · KITALAR · RENK PALETİ ---- */
  function paintForces() {
    var F = state.forces;
    if (!F) return;
    var A = (F.armies || [])[0], B = (F.armies || [])[1];
    if (A && $('#armyA')) {
      $('#armyA').querySelector('b').textContent = A.name;
      $('#armyA').querySelector('i').style.background = A.hue;
    }
    if (B && $('#armyB')) {
      $('#armyB').querySelector('b').textContent = B.name;
      $('#armyB').querySelector('i').style.background = B.hue;
    }
    var cr = $('#corpsRow');
    if (cr && F.corps) {
      cr.innerHTML = '';
      F.corps.forEach(function (c) {
        var d = document.createElement('div');
        d.className = 'cq';
        d.title = c.name + ' · ağırlık ' + c.weight + (c.far ? ' · cephe gerisi' : '');
        d.style.opacity = c.far ? 0.45 : 0.9;
        cr.appendChild(d);
      });
    }
    var pr = $('#palRow');
    if (pr && F.palette) {
      pr.innerHTML = '';
      var names = { str: 'yaylı', brass: 'bakır', perc: 'vurmalı', wood: 'tahta', choir: 'koro', bell: 'çan' };
      Object.keys(F.palette.w).forEach(function (k) {
        var sp = document.createElement('span');
        sp.textContent = names[k] + ' ' + F.palette.w[k].toFixed(2);
        if (F.palette.hi === k) sp.className = 'hi';
        pr.appendChild(sp);
      });
    }
  }

  /* kıtaların ağırlığını ölçü ölçü canlandır (statik değil) */
  function paintCorps(bar) {
    var cr = $('#corpsRow');
    if (!cr || !state.score || !state.score.forces) return;
    var corps = state.score.forces.corps, kids = cr.children;
    var fp = state.score.forces;
    for (var i = 0; i < kids.length && i < corps.length; i++) {
      var ph = (i * 1.7) + bar * (0.11 + (i % 3) * 0.05);
      var w = Math.max(0.10, corps[i].weight * (0.55 + 0.75 * (0.5 + 0.5 * Math.sin(ph))));
      kids[i].style.height = Math.max(2, Math.min(16, w * 12)).toFixed(0) + 'px';
      kids[i].style.opacity = corps[i].far ? (0.25 + w * 0.35).toFixed(2) : (0.45 + w * 0.45).toFixed(2);
    }
  }

    var box = $('#kimlik');

    var no = sy ? sy.total() : 0;

      ['ESER', '#' + no + ' · ' + k.aranjman + ' · koro ' + k.koro],
      ['ORDULAR', (k.ordular || []).join('  vs  ') + '  ·  ' + k.orduRenk],
      ['DURUMLAR', (k.durumlar || []).join(' → ')],
      ['TEMPOLAR', (k.tempoHarita || []).join('  ·  ')],
      ['DUYGULAR', Object.keys(k.duygular || {}).map(function (x) { return x + ' ' + k.duygular[x] + ' ölçü'; }).join(' · ')],
      ['MOTİF', (k.motifHucresi || '') + '   EZGİ ' + (k.ezgiNotasi || '')],
      ['OSTİNATO', k.ostinato || '—'],
      ['φ KAFES', k.fibonacci + '   altın kesit ölçü ' + k.altinKesit],
      ['KAOS', k.kaosFaz + ' faz · ateşkes ' + k.ateskes + ' · vurmalı ' + (k.vurmali ? k.vurmali.bars : 0) + ' ölçü'],
      ['JEST', k.jest ? ('yükseliş ' + k.jest.yukselis + ' · çöküş ' + k.jest.cokus + ' · çığlık ' + k.jest.ciglik + ' · galeyan ' + k.jest.galeyan + ' · yalvarma ' + k.jest.yalvarma) : '—']
    ];
    var html = satir.map(function (r) {
      return '<div class="kr"><span>' + r[0] + '</span><b>' + r[1] + '</b></div>';
    }).join('');
    if (state.benz) {
      html += '<div class="kr kr-b"><span>BENZERSİZLİK</span><b>' +
        (state.score.benzersiz === false ? 'TEKRAR!' : 'YENİ') +
        ' · en yakın benzerlik %' + (state.benz.max * 100).toFixed(0) +
        ' · arşiv ' + state.benz.n + '</b></div>';
    }
    box.innerHTML = html;
  }

  /* ---- DURUM & DUYGU ŞERİDİ: her ölçünün cephe durumu ve duygusu ---- */
  function paintEmotion(bar) {
    var box = $('#emoStrip');
    if (!box || !state.score || !state.score.emotions) return;
    var em = state.score.emotions;
    if (!state._emoCells || state._emoCells.length !== em.length) {
      box.innerHTML = '';
      state._emoCells = [];
      em.forEach(function (e, i) {
        var d = document.createElement('i');
        d.title = (i + 1) + '. ölçü · ' + e.phase + ' · ' + e.emo;
        box.appendChild(d);
        state._emoCells.push(d);
      });
    }
    var COL = { savas: '#c0392b', zafer: '#f2d79a', yenilgi: '#5b6b8a', direnc: '#7fd8c8',
      ofke: '#e0607a', uyum: '#8fb6e0' };
    for (var i = 0; i < em.length; i++) {
      var cell = state._emoCells[i];
      if (!cell) continue;
      var on = i <= bar;
      cell.style.background = COL[em[i].emo] || '#666';
      cell.style.opacity = on ? (0.45 + Math.min(0.55, em[i].dyn * 0.45)).toFixed(2) : '0.10';
      cell.style.transform = 'scaleY(' + (on ? (0.30 + Math.min(1, em[i].dens) * 0.7).toFixed(2) : '0.25') + ')';
    }
  }

  /* ---- CEPHE HATTI: savaşın gidişatı ---- */
  function paintFront(bar) {
    var cv = $('#front');
    if (!cv || !cv.getContext || !state.battle) return;
    var c = cv.getContext('2d');
    var W = cv.width, H = cv.height;
    c.clearRect(0, 0, W, H);
    var map = state.battle.map;
    var mid = H / 2;
    /* denge eğrisi */
    for (var i = 1; i < map.length; i++) {
      var x0 = ((i - 1) / map.length) * W, x1 = (i / map.length) * W;
      var y0 = mid - map[i - 1].b * (H * 0.42), y1 = mid - map[i].b * (H * 0.42);
      var cur = i <= bar;
      c.strokeStyle = cur ? 'rgba(242,215,154,.95)' : 'rgba(160,170,190,.28)';
      c.lineWidth = cur ? 2 : 1.2;
      c.beginPath(); c.moveTo(x0, y0); c.lineTo(x1, y1); c.stroke();
    }
    /* orta çizgi (eşit güçler) */
    c.strokeStyle = 'rgba(255,255,255,.12)'; c.lineWidth = 1;
    c.beginPath(); c.moveTo(0, mid); c.lineTo(W, mid); c.stroke();
    /* kıtalar: sol / merkez / sağ */
    var bi = map[Math.min(map.length - 1, Math.max(0, bar))] || { b: 0 };
    ['solKita', 'sagKita', 'basHat', 'oncu', 'yedek'].forEach(function (k) {
      var R = root.Battle.REG[k];
      var px = R.x + bi.b * 0.30 * (R.x >= 0 ? 1 : -1);
      var pz = R.z + bi.b * 0.35;
      var x = W * (0.5 + px * 0.42);
      var y = mid + (pz * H * 0.34);
      var sz = 5 + Math.max(0, R.y) * 10;
      c.beginPath();
      c.fillStyle = k === 'solKita' ? 'rgba(127,216,200,.9)' : k === 'sagKita' ? 'rgba(242,215,154,.9)'
        : k === 'oncu' ? 'rgba(226,96,122,.9)' : k === 'yedek' ? 'rgba(143,182,224,.75)' : 'rgba(200,205,220,.7)';
      c.arc(x, y, sz, 0, Math.PI * 2); c.fill();
    });
    /* imleç */
    var px2 = (Math.min(bar, map.length) / map.length) * W;
    c.fillStyle = 'rgba(255,255,255,.85)'; c.fillRect(px2, 0, 1.4, H);
    /* faz adı */
    var cur = map[Math.min(map.length - 1, Math.max(0, bar))];
    if ($('#phaseName') && cur) {
      $('#phaseName').textContent = cur.phaseName;
      $('#frontVal').textContent = (cur.b > 0 ? '+' : '') + cur.b.toFixed(2);
      var dots = document.querySelectorAll('.phase-dot');
      for (var d = 0; d < dots.length; d++) dots[d].classList.toggle('on', dots[d].getAttribute('data-p') === cur.phase);
    }
  }

  function paintSection() {
    var sc0 = state.score, sec0 = sc0 && sc0.sections ? sc0.sections[state.secIdx] : null;
    if (sec0 && $('#secPhase')) {
      $('#secPhase').textContent = (sec0.phaseName || '—');
      $('#secBpm').textContent = (sec0.bpm != null ? sec0.bpm + ' BPM' : '—');
    }

    var sc = state.score, sec = sc.sections[state.secIdx];
    if (!sec) return;
    if ($('#secName')) $('#secName').textContent = sec.name;
    if ($('#secDyn')) $('#secDyn').textContent = sec.dyn;
    if ($('#secBars')) $('#secBars').textContent = sec.bars + ' ölçü';
    var dots = document.querySelectorAll('.sec-dot');
    for (var i = 0; i < dots.length; i++) dots[i].classList.toggle('on', i === state.secIdx);
    if ($('#scoreSheet')) renderSheet();
  }

  function drawTimeline(elapsed) {
    var cv = $('#tl');
    if (!cv || !cv.getContext) return;
    var ctx = cv.getContext('2d');
    var W = cv.width, H = cv.height;
    ctx.clearRect(0, 0, W, H);
    var sc = state.score;
    /* bölüm şeritleri */
    sc.sections.forEach(function (s, i) {
      var x = (s.start / sc.totalSec) * W, w = (s.dur / sc.totalSec) * W;
      var on = i === state.secIdx;
      ctx.fillStyle = on ? 'rgba(214,178,110,.30)' : 'rgba(255,255,255,.055)';
      ctx.fillRect(x + 1, 0, w - 2, H);
      ctx.fillStyle = on ? 'rgba(245,215,150,.95)' : 'rgba(190,196,210,.55)';
      ctx.font = '10px ui-monospace, monospace';
      ctx.fillText(s.name.toUpperCase(), x + 6, 12);
    });
    /* --- ATEŞKES / SUS pencereleri: koyu boşluk bantları --- */
    if (sc.ceasefire && sc.ceasefire.detail) {
      sc.ceasefire.detail.forEach(function (d) {
        var x0 = (d.start * sc.bar / sc.totalSec) * W;
        var x1 = ((d.start + d.len) * sc.bar / sc.totalSec) * W;
        ctx.fillStyle = 'rgba(8,10,16,0.82)';
        ctx.fillRect(x0, 0, Math.max(2, x1 - x0), H);
        ctx.strokeStyle = 'rgba(120,200,190,0.55)';
        ctx.lineWidth = 1;
        ctx.beginPath(); ctx.moveTo(x0, H - 2); ctx.lineTo(x1, H - 2); ctx.stroke();
        ctx.fillStyle = 'rgba(140,220,205,0.85)';
        ctx.font = '8px ui-monospace, monospace';
        ctx.fillText(d.kind.toUpperCase(), x0 + 3, H - 5);
      });
    }
    /* --- JEST işaretleri: çığlık / galeyan / yalvarma / tümsek --- */
    if (sc.jest) {
      var jm = sc.jest;
      function mark(bar, color, h) {
        var x = (bar * sc.bar / sc.totalSec) * W;
        ctx.fillStyle = color;
        ctx.fillRect(x, 0, 2, h);
      }
      if (jm.rises) mark(0, '', 0);
      /* plan pencereleri skorda tutulmuyor; jest sayılarından yoğunluk şeridi */
      ctx.fillStyle = 'rgba(226,96,122,0.20)';
      ctx.fillRect(0, 0, W, 2);
    }

    /* --- KAOS PENCERELERİ: rastlantısal mix fazları (kaotik bant) --- */
    if (sc.chaos && sc.chaos.detail) {
      sc.chaos.detail.forEach(function (d) {
        var x0 = (d.start * sc.bar / sc.totalSec) * W;
        var x1 = ((d.start + d.len) * sc.bar / sc.totalSec) * W;
        var g = ctx.createLinearGradient(x0, 0, x1, 0);
        g.addColorStop(0, 'rgba(226,96,122,0.05)');
        g.addColorStop(0.5, 'rgba(226,96,122,' + (0.16 + d.amp * 0.16).toFixed(2) + ')');
        g.addColorStop(1, 'rgba(248,214,140,0.06)');
        ctx.fillStyle = g;
        ctx.fillRect(x0, H - 9, Math.max(2, x1 - x0), 7);
        /* karışıklık dalgası */
        ctx.strokeStyle = 'rgba(255,190,150,0.45)';
        ctx.lineWidth = 1;
        ctx.beginPath();
        var seg = Math.max(4, Math.floor((x1 - x0) / 3));
        for (var k = 0; k <= seg; k++) {
          var xx = x0 + (x1 - x0) * (k / seg);
          var yy = H - 13 - Math.sin(k * 1.9 + d.amp * 6) * (2 + d.amp * 3);
          if (k === 0) ctx.moveTo(xx, yy); else ctx.lineTo(xx, yy);
        }
        ctx.stroke();
      });
    }
    /* olay yoğunluğu (nota tülü) */
    var ev = sc.events;
    for (var i = 0; i < ev.length; i += 3) {
      var e = ev[i];
      var ex = (e.t / sc.totalSec) * W;
      var lane = Math.max(0, Math.min(H - 1, H - 6 - ((e.midi - 30) / 80) * (H - 14)));
      ctx.fillStyle = e.t <= elapsed ? 'rgba(130,220,210,.55)' : 'rgba(255,255,255,.10)';
      ctx.fillRect(ex, lane, 1.6, 1.6);
    }
    /* imleç */
    var px = (Math.min(elapsed, sc.totalSec) / sc.totalSec) * W;
    ctx.fillStyle = 'rgba(255,225,160,.95)';
    ctx.fillRect(px, 0, 1.5, H);
  }

  /* ---- nota kağıdı (partisyon görünümü) ---- */
  function renderSheet() {
    var host = $('#scoreSheet');
    if (!host) return;
    var sc = state.score;
    var html = '';
    var rows = ['İlk Keman', '2. Keman', 'Viyola', 'Çello', 'Kontrbas',
      'Korno', 'Trompet', 'Trombon/Tuba', 'Timpali · Davul', 'Zil · Gong'];
    var map = {
      'İlk Keman': ['violin', 'glockenspiel'], '2. Keman': ['string_ensemble_1', 'flute'],
      'Viyola': ['viola', 'oboe'], 'Çello': ['cello', 'clarinet'], 'Kontrbas': ['contrabass', 'tuba', 'pizzicato_strings'],
      'Korno': ['french_horn'], 'Trompet': ['trumpet', 'brass_section'], 'Trombon/Tuba': ['trombone', 'tuba', 'bassoon'],
      'Timpali · Davul': ['timpani', 'taiko_drum', 'woodblock'], 'Zil · Gong': ['reverse_cymbal', 'orchestra_hit', 'tubular_bells']
    };
    rows.forEach(function (label) {
      var insts = map[label];
      var cells = '';
      var NCELL = 64;
      var buckets = new Array(NCELL).fill(0);
      sc.events.forEach(function (e) {
        if (insts.indexOf(e.inst) < 0) return;
        var k = Math.floor((e.t / sc.totalSec) * NCELL);
        if (k >= 0 && k < NCELL) buckets[k] += e.v;
      });
      var mx = Math.max(0.001, Math.max.apply(null, buckets));
      for (var i = 0; i < NCELL; i++) {
        var a = buckets[i] / mx;
        cells += '<i style="opacity:' + (0.06 + a * 0.94).toFixed(3) + '"></i>';
      }
      html += '<div class="sh-row"><b>' + label + '</b><div class="sh-cells">' + cells + '</div></div>';
    });
    host.innerHTML = html;
  }

  function flashLoop() {
    var el = $('#loopFlash');
    if (!el) return;
    el.classList.add('show');
    setTimeout(function () { el.classList.remove('show'); }, 1400);
  }

  /* ---- kontrol ---- */
  function start() {
    A.init();
    A.resume();
    if (!A.isReady()) return;
    A.fadeIn(0.5);
    state.playing = true;
    if (!state.score) newScore(true);
    if (state.timer) clearInterval(state.timer);
    state.timer = setInterval(schedule, 90);
    schedule();
    if ($('#playBtn')) $('#playBtn').textContent = '❚❚ DURAKLAT';
    document.body.classList.add('playing');
  }
  function stop() {
    state.playing = false;
    if (state.timer) { clearInterval(state.timer); state.timer = null; }
    A.fadeOut(0.45);

    if ($('#playBtn')) $('#playBtn').textContent = '▶ ÇAL';
    document.body.classList.remove('playing');
  }
  function toggle() { state.playing ? stop() : start(); }

  function boot() {
    A.init();
    var bar = $('#loadBar'), lab = $('#loadLabel');
    A.loadAll(function (done, total, inst) {
      if (bar) bar.style.width = Math.round((done / total) * 100) + '%';
      if (lab) lab.textContent = 'orkestra akortlanıyor · ' + inst + ' (' + done + '/' + total + ')';
    }).then(function () {
      if (lab) lab.textContent = 'orkestra hazır · gerçek kayıtlı örnekler (' + A.instList().length + ' enstrüman)';
