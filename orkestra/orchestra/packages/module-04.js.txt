  ST.decorate = function (melody, motif, R, rootMidi) {
    if (!melody || !melody.length) return melody;
    var out = [], m = 0, i = 0;
    while (i < melody.length) {
      out.push(melody[i]);
      if (R() < 0.45 && melody[i][1] >= 1) {
        /* uzun notanın içine motif hücresi */
        var mm = motif[m % motif.length];
        for (var k = 0; k < mm.length; k++) {
          out.push([rootMidi + mm[k], 0.25]);
        }
        m++;
      }
      i++;
    }
    return out;
  };
